// ============ PARSER OPSI MENU (format sama dengan Manajemen Produk) ============
// "Pedas:Level1,Level2 | Topping:Keju+2000,Telur,Bakso[maks2]"
function parseOpsiMenu(opsiStr) {
  opsiStr = (opsiStr || '').trim();
  if (!opsiStr) return [];
  var RADIO_KEYWORDS = ['pedas','level','ukuran','size','panas','suhu','jenis','pilih'];

  return opsiStr.split('|').map(function(bagian) {
    bagian = bagian.trim();
    if (!bagian) return null;

    var maks = null;
    var maksMatch = bagian.match(/\[maks(\d+)\]/i);
    if (maksMatch) { maks = parseInt(maksMatch[1]); bagian = bagian.replace(/\[maks\d+\]/i, '').trim(); }

    var colonIdx = bagian.indexOf(':');
    if (colonIdx < 0) return null;
    var namaGrup = bagian.substring(0, colonIdx).trim();
    var pilihanStr = bagian.substring(colonIdx + 1).trim();

    var pilihan = pilihanStr.split(',').map(function(p) {
      p = p.trim();
      var hargaMatch = p.match(/\+(\d+)$/);
      var harga = hargaMatch ? parseInt(hargaMatch[1]) : 0;
      var nama = hargaMatch ? p.replace(/\+\d+$/, '').trim() : p;
      return { nama: nama, harga: harga };
    }).filter(function(p){ return p.nama; });
    if (!pilihan.length) return null;

    var namaLower = namaGrup.toLowerCase();
    var isRadio = maks === 1 || RADIO_KEYWORDS.some(function(k){ return namaLower.indexOf(k) >= 0; });

    if (isRadio) {
      return { group: namaGrup, sub: 'Wajib pilih 1', type: 'radio', max: 1, items: pilihan };
    }
    var maxPilih = maks || pilihan.length;
    return { group: namaGrup, sub: maks ? 'Pilih maks. ' + maks : 'Pilih sesuka hati (opsional)', type: 'checkbox', max: maxPilih, items: pilihan };
  }).filter(Boolean);
}

function namaOpsiNonaktif(opsiNonaktifStr) {
  return String(opsiNonaktifStr || '').split(',').map(function(x){ return x.trim().toLowerCase(); }).filter(Boolean);
}

function renderBanner(list) {
  var wrap = document.getElementById('bannerWrap');
  var track = document.getElementById('bannerTrack');
  var dots = document.getElementById('bannerDots');
  if (!wrap || !track || !dots) return;
  if (_bannerTimer) { clearInterval(_bannerTimer); _bannerTimer = null; }
  list = list.filter(function(u) { return /^https:\/\//i.test(String(u || '')); });
  if (!list.length) { wrap.classList.remove('show'); track.innerHTML = ''; dots.innerHTML = ''; return; }
  track.innerHTML = list.map(function(u, i) {
    return '<div class="banner-slide"><img src="' + String(u).replace(/"/g, '&quot;') + '" alt="Banner ' + (i + 1) + '" draggable="false"' + (i === 0 ? '' : ' loading="lazy"') + '></div>';
  }).join('');
  dots.innerHTML = list.length > 1 ? list.map(function(u, i) {
    return '<button type="button" class="' + (i === 0 ? 'on' : '') + '" aria-label="Banner ' + (i + 1) + '" onclick="keBanner(' + i + ')"></button>';
  }).join('') : '';
  track.scrollLeft = 0;
  wrap.classList.add('show');
  // gambar yang gagal dimuat disembunyikan, banner lain tetap tampil
  Array.prototype.forEach.call(track.querySelectorAll('img'), function(im) {
    im.onerror = function() { var s = im.parentNode; if (s && s.parentNode) s.parentNode.removeChild(s); var sisa = track.querySelectorAll('.banner-slide').length; if (!sisa) wrap.classList.remove('show'); };
  });
  track.onscroll = function() {
    var idx = Math.round(track.scrollLeft / (track.clientWidth || 1));
    var ds = dots.querySelectorAll('button');
    for (var k = 0; k < ds.length; k++) ds[k].classList.toggle('on', k === idx);
  };
  // jeda putar-otomatis saat pelanggan menyentuh / menggeser sendiri
  var jeda = function() { _bannerJeda = Date.now() + 8000; };
  track.ontouchstart = jeda; track.onpointerdown = jeda; track.onwheel = jeda;
  if (list.length > 1) {
    _bannerTimer = setInterval(function() {
      if (Date.now() < _bannerJeda || document.hidden) return;
      var n = track.querySelectorAll('.banner-slide').length;
      if (n < 2) return;
      var idx = Math.round(track.scrollLeft / (track.clientWidth || 1));
      keBanner((idx + 1) % n);
    }, 5000);
  }
}

// ============ RIWAYAT PESANAN & AKTIVITAS (cache browser) ============
// Riwayat pesanan yang berhasil dikirim, plus data pemesan terakhir (nama & alamat), disimpan di
// localStorage perangkat pelanggan — tetap ada setelah halaman ditutup/dimuat ulang. Foto cetak
// TIDAK disimpan (terlalu besar), hanya jumlahnya. Jika localStorage diblokir browser, data
// disimpan sementara di memori selama halaman terbuka agar fitur tidak error.
var RIWAYAT_KEY = 'menuPelangganRiwayatV1';
var PEMESAN_KEY = 'menuPelangganPemesanV1';
var RIWAYAT_MAKS = 30;
var _memCache = {};

function _cacheGet(k) {
  try { var v = window.localStorage.getItem(k); if (v !== null) return v; } catch (e) {}
  return (_memCache[k] !== undefined) ? _memCache[k] : null;
}
function _cacheSet(k, v) {
  _memCache[k] = v;
  try { window.localStorage.setItem(k, v); } catch (e) {}
}

function muatRiwayatPesanan() {
  try {
    var arr = JSON.parse(_cacheGet(RIWAYAT_KEY) || '[]');
    return Array.isArray(arr) ? arr.filter(function(r) { return r && r.id; }) : [];
  } catch (e) { return []; }
}

function catatRiwayatPesanan(txId, nama, alamatV, catatan, total) {
  try {
    var entri = {
      id: String(txId),
      waktu: Date.now(),
      nama: String(nama || ''),
      total: Number(total) || 0,
      alamat: alamatRingkas(alamatV),
      catatan: String(catatan || ''),
      items: cart.map(function(c) {
        return {
          id: c.menuId,
          nama: String(c.nama || ''),
          opsi: (c.opsiTerpilih || []).map(function(o) { return o.nama; }),
          qty: Number(c.qty) || 1,
          harga: Number(c.hargaSatuan) || 0,
          catatan: String(c.catatan || ''),
          foto: (c.fotos && c.fotos.length) || 0
        };
      })
    };
    var daftar = muatRiwayatPesanan().filter(function(r) { return r.id !== entri.id; });
    daftar.unshift(entri);
    _cacheSet(RIWAYAT_KEY, JSON.stringify(daftar.slice(0, RIWAYAT_MAKS)));
    _cacheSet(PEMESAN_KEY, JSON.stringify({ nama: entri.nama, alamat: alamatV }));
  } catch (e) {}
}

// "Pesan Lagi": buka halaman Detail Produk dari item yang pernah dipesan.
// Produk dicari berdasarkan ID; untuk riwayat lama (tanpa ID) dicocokkan lewat nama produk.
function pesanLagiRiwayat(i, j) {
  var r = muatRiwayatPesanan()[i];
  var it = r && r.items && r.items[j];
  if (!it) return;
  var idx = -1, k;
  if (it.id !== undefined && it.id !== null) {
    for (k = 0; k < menus.length; k++) { if (String(menus[k].id) === String(it.id)) { idx = k; break; } }
  }
  if (idx < 0) {
    var nm = String(it.nama || '').trim().toLowerCase();
    for (k = 0; k < menus.length; k++) { if (String(menus[k].nama || '').trim().toLowerCase() === nm) { idx = k; break; } }
  }
  if (idx < 0) { showToast('Produk ini sudah tidak tersedia di menu'); return; }
  tutupSheet('riwayatPageBg');
  bukaOpsiMenu(idx);
}

function hapusRiwayatPesanan(i) {
  var daftar = muatRiwayatPesanan();

function salinTeks(t) {
  function fallback() {
    var ok = false;
    try {
      var ta = document.createElement('textarea');
      ta.value = t;
      ta.setAttribute('readonly', '');
      ta.style.position = 'fixed'; ta.style.opacity = '0'; ta.style.left = '-9999px';
      document.body.appendChild(ta);
      ta.select();
      ta.setSelectionRange(0, t.length);
      ok = document.execCommand('copy');
      document.body.removeChild(ta);
    } catch (e) {}
    return ok;
  }
  try {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(t).catch(function() { fallback(); });
      return true;
    }
  } catch (e) {}
  return fallback();
}

function _hexKeRgb(hex) {
  var m = /^#?([0-9a-f]{2})([0-9a-f]{2})([0-9a-f]{2})$/i.exec(String(hex || ''));
  if (!m) return [255, 255, 255];
  return [parseInt(m[1], 16), parseInt(m[2], 16), parseInt(m[3], 16)];
}

function renderSampul(res) {
  var el = document.getElementById('cover');
  if (!el) return;
  var s = res && res.success ? res.sampul : null;
  if (!s || !s.aktif) { tutupSampul(true); return; }
  var esc = function(v) { return String(v == null ? '' : v).replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); };
  var rgb = _hexKeRgb(s.warna);
  var lum = (0.299 * rgb[0] + 0.587 * rgb[1] + 0.114 * rgb[2]) / 255;
  el.style.setProperty('--aksen', s.warna || '#FFFFFF');
  el.style.setProperty('--aksen-rgb', rgb.join(','));
  el.style.setProperty('--aksen-teks', lum > 0.6 ? '#111111' : '#FFFFFF');
  el.style.setProperty('--gelap', String(typeof s.gelap === 'number' ? s.gelap : 0.4));

  document.getElementById('coverSapaan').textContent = s.sapaan || '';
  document.getElementById('coverJudul').textContent = s.judul || res.nama || '';
  document.getElementById('coverKalimat').textContent = s.kalimat || '';
  document.getElementById('coverBtnTeks').textContent = s.tombol || 'Lihat Menu';

  var logoBox = document.getElementById('coverLogo');
  if (s.tampilLogo && /^https:\/\//i.test(String(res.logo || ''))) {
    logoBox.innerHTML = '<img src="' + esc(res.logo) + '" alt="Logo" draggable="false">';
    var li = logoBox.querySelector('img');
    if (li) li.onerror = function() { logoBox.classList.remove('show'); logoBox.innerHTML = ''; };
    logoBox.classList.add('show');
  }

  var gambar = (s.gambar || []).filter(function(u) { return /^https:\/\//i.test(String(u || '')); });
  var slides = document.getElementById('coverSlides');
  var dots = document.getElementById('coverDots');
  slides.innerHTML = gambar.map(function(u) {
    return '<div class="cover-slide" style="background-image:url(\'' + esc(u) + '\')"></div>';
  }).join('');
  dots.innerHTML = gambar.length > 1 ? gambar.map(function() { return '<span></span>'; }).join('') : '';

  var tampil = function() {
    if (el._siap) return; el._siap = true;
    var sl = slides.querySelectorAll('.cover-slide'), dt = dots.querySelectorAll('span');
    var idx = 0;
    var pilih = function(i) {
      for (var k = 0; k < sl.length; k++) sl[k].classList.toggle('on', k === i);
      for (var j = 0; j < dt.length; j++) dt[j].classList.toggle('on', j === i);
    };
    if (sl.length) pilih(0);
    if (sl.length > 1) {
      _sampulTimer = setInterval(function() { if (document.hidden) return; idx = (idx + 1) % sl.length; pilih(idx); }, 5000);
    }
    el.classList.add('ready');
    var b = document.getElementById('coverBtn'); if (b) try { b.focus({ preventScroll: true }); } catch (e) {}
  };
  if (gambar.length) {
    // tunggu gambar pertama termuat agar tidak berkedip (maks. 4 dtk, gagal pun tetap lanjut)
    var im = new Image(), t = setTimeout(tampil, 4000);
    im.onload = im.onerror = function() { clearTimeout(t); tampil(); };
    im.src = gambar[0];
  } else {
    tampil();
  }
}

function tutupSampul(langsung) {
  var el = document.getElementById('cover');
  document.documentElement.classList.remove('cover-open');
  if (_sampulTimer) { clearInterval(_sampulTimer); _sampulTimer = null; }
  if (!el) return;
  var hapus = function() { if (el.parentNode) el.parentNode.removeChild(el); };
  if (langsung) { hapus(); return; }
  el.classList.add('leaving');
  setTimeout(hapus, 800);
}

            })(),
            tersedia: item.tersedia !== false

// Label kategori di atas nama produk & tombol "Pilih Produk" pada kartu (kartu tetap bisa diklik penuh).
var KARTU_BTN_HTML = '<div class="btn-pilih">Pilih Produk</div>';
// Harga coret (diatur di Manajemen Produk). Hanya tampil bila hargaCoret > harga jual; selain itu tampilan harga tetap seperti semula.
function adaHargaCoret(m) { return !!(m && m.hargaCoret > 0 && m.harga > 0 && m.hargaCoret > m.harga); }
function hargaIsiHtml(m, denganBadge) {
  var now = 'Rp ' + m.harga.toLocaleString('id-ID');
  if (!adaHargaCoret(m)) return now;
  var persen = Math.round((m.hargaCoret - m.harga) / m.hargaCoret * 100);
  return '<span class="harga-atas"><span class="harga-coret">Rp ' + m.hargaCoret.toLocaleString('id-ID') + '</span>' +
    (denganBadge && persen > 0 ? '<span class="harga-diskon">-' + persen + '%</span>' : '') + '</span>' +
    '<span class="harga-now">' + now + '</span>';
}
function hargaDivHtml(m) {
  return '<div class="harga' + (adaHargaCoret(m) ? ' ada-coret' : '') + '">' + hargaIsiHtml(m, false) + '</div>';
}
function kartuBrandHtml(m) {
  var k = String((m && m.kat) || '').replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  return k ? '<div class="brand">' + k + '</div>' : '';
}

function rekomKartuHtml(m) {
  var esc = function(v) { return String(v == null ? '' : v).replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); };
  var idx = menus.indexOf(m);
  var foto = m.img || (m.imgs && m.imgs[0]) || '';
  var imgHtml = foto
    ? '<img src="' + esc(foto) + '" alt="' + esc(m.nama) + '" loading="lazy" draggable="false" onerror="this.onerror=null;this.style.display=\'none\';">'
    : '<div class="ph">🍽️</div>';
  return '<div class="menu-card' + (!m.tersedia ? ' habis' : '') + '" onclick="bukaOpsiMenu(' + idx + ')">' +
    (!m.tersedia ? '<div class="habis-flag">Habis</div>' : '') +
    '<div class="img-box">' + imgHtml + '</div>' +
    '<div class="info">' + kartuBrandHtml(m) + '<div class="nama">' + esc(m.nama) + '</div>' +
    hargaDivHtml(m) + '</div>' + KARTU_BTN_HTML +
  '</div>';
}

function isiRekomendasi(wrapId, trackId, kecualiId) {
  var wrap = document.getElementById(wrapId), track = document.getElementById(trackId);
  if (!wrap || !track) return;
  var list = menus.filter(function(m) { return m.rekomendasi && (kecualiId == null || String(m.id) !== String(kecualiId)); });
  if (!list.length) { wrap.classList.remove('show'); track.innerHTML = ''; return; }
  track.innerHTML = list.map(rekomKartuHtml).join('');
  track.scrollLeft = 0;
  wrap.classList.add('show');
}

// ============ PRODUK BARU (maks. 3 produk terakhir ditambahkan, tampil di bawah Rekomendasi) ============
// "Terakhir ditambahkan" = ID terbesar (ID baru selalu maxId+1 saat produk ditambah). Hanya produk yang tersedia.
// Bagian disembunyikan otomatis jika belum ada produk.
var _baruTimer = null, _baruIdx = 0, _baruJeda = false, _baruBound = false;
function _baruBerhenti() { if (_baruTimer) { clearInterval(_baruTimer); _baruTimer = null; } }
function _baruKe(i, halus) {
  var list = document.getElementById('baruMenuList');
  if (!list) return;
  var n = list.children.length;
  if (!n) return;
  i = ((i % n) + n) % n;
  _baruIdx = i;
  list.scrollTo({ left: i * list.clientWidth, behavior: halus === false ? 'auto' : 'smooth' });
  _baruTandaiDot(i);
}
function _baruTandaiDot(i) {
  var d = document.getElementById('baruMenuDots');
  if (!d) return;
  for (var k = 0; k < d.children.length; k++) d.children[k].classList.toggle('on', k === i);
}
function _baruMulai() {
  _baruBerhenti();
  var list = document.getElementById('baruMenuList');
  if (!list || list.children.length < 2) return;
  if (window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  _baruTimer = setInterval(function() {
    if (_baruJeda || document.hidden) return;
    _baruKe(_baruIdx + 1);
  }, 4000);
}
function _baruPasangEvent() {
  if (_baruBound) return;
  var list = document.getElementById('baruMenuList');
  if (!list) return;
  _baruBound = true;
  var lanjutT = null;
  var jeda = function() { _baruJeda = true; if (lanjutT) clearTimeout(lanjutT); };
  var lanjut = function(tunda) { if (lanjutT) clearTimeout(lanjutT); lanjutT = setTimeout(function() { _baruJeda = false; }, tunda); };
  list.addEventListener('touchstart', jeda, { passive: true });
  list.addEventListener('touchend', function() { lanjut(3000); }, { passive: true });
  list.addEventListener('touchcancel', function() { lanjut(3000); }, { passive: true });
  list.addEventListener('mouseenter', jeda);
  list.addEventListener('mouseleave', function() { lanjut(800); });
  // Sinkronkan titik indikator saat digeser manual / otomatis
  var st = null;
  list.addEventListener('scroll', function() {
    if (st) clearTimeout(st);
    st = setTimeout(function() {
      var w = list.clientWidth || 1;
      var i = Math.round(list.scrollLeft / w);
      _baruIdx = i; _baruTandaiDot(i);
    }, 60);
  }, { passive: true });
  // Rapikan posisi bila ukuran layar berubah (rotasi HP, dsb.)
  window.addEventListener('resize', function() { _baruKe(_baruIdx, false); });
}
function renderBaruMenu() {
  var wrap = document.getElementById('baruMenu'), list = document.getElementById('baruMenuList');
  var dots = document.getElementById('baruMenuDots');
  if (!wrap || !list) return;
  var esc = function(v) { return String(v == null ? '' : v).replace(/&/g, '&amp;').replace(/"/g, '&quot;').replace(/</g, '&lt;').replace(/>/g, '&gt;'); };
  var urut = menus.map(function(m, i) {
    var n = Number(m.id);
    return { m: m, i: i, k: (String(m.id == null ? '' : m.id).trim() !== '' && isFinite(n)) ? n : i };
  }).filter(function(o) { return o.m.tersedia; })
    .sort(function(a, b) { return (b.k - a.k) || (b.i - a.i); })
    .slice(0, 3);
  if (!urut.length) { _baruBerhenti(); wrap.classList.remove('show'); list.innerHTML = ''; if (dots) { dots.innerHTML = ''; dots.classList.remove('show'); } return; }
  list.innerHTML = urut.map(function(o) {
    var m = o.m;
    var foto = m.img || (m.imgs && m.imgs[0]) || '';
    var imgHtml = foto
      ? '<img src="' + esc(foto) + '" alt="' + esc(m.nama) + '" loading="lazy" draggable="false" onerror="this.onerror=null;this.style.display=\'none\';">'
      : '<div class="ph">🍽️</div>';
    return '<div class="baru-slide"><div class="baru-card" onclick="bukaOpsiMenu(' + o.i + ')">' +
      '<div class="baru-ribbon">BARU</div>' +
      '<div class="baru-img">' + imgHtml + '</div>' +
      '<div class="baru-info">' +
        (m.kat ? '<div class="baru-kat">' + esc(m.kat) + '</div>' : '') +
        '<div class="baru-nama">' + esc(m.nama) + '</div>' +
        hargaDivHtml(m) +
      '</div>' +
      '<div class="baru-go">›</div>' +
    '</div></div>';
  }).join('');
  wrap.classList.add('show');
  // Titik indikator + geser otomatis (hanya bila ada lebih dari 1 produk)
  _baruIdx = 0;
  list.scrollLeft = 0;
  if (dots) {
    if (urut.length > 1) {
      dots.innerHTML = urut.map(function(_, k) { return '<button type="button" aria-label="Produk baru ' + (k + 1) + '" onclick="_baruKe(' + k + ');_baruJeda=false;"' + (k === 0 ? ' class="on"' : '') + '></button>'; }).join('');
      dots.classList.add('show');
    } else { dots.innerHTML = ''; dots.classList.remove('show'); }
  }
  _baruPasangEvent();
  _baruMulai();
}

// ============ KATEGORI POPULER (produk terlaris 1–5) ============
// Diambil dari data penjualan Kasir (Detail_Transaksi) lewat getProdukPopuler().
// Jika belum ada penjualan, tab Populer tidak ditampilkan.
var _populerIds = [];
var _terjual = {};   // id produk → jumlah terjual (untuk mengurutkan produk terlaris di tiap kategori)

function urutanTerlaris(list) {
  // Produk terlaris di urutan pertama; yang jumlah terjualnya sama tetap mengikuti urutan menu semula.
  return list.slice().sort(function(a, b) {
    var d = (_terjual[String(b.id)] || 0) - (_terjual[String(a.id)] || 0);
    return d || (menus.indexOf(a) - menus.indexOf(b));
  });
}

function muatPopuler(percobaan) {
  tambahTabPopuler();   // tab Populer langsung tampil; isinya menyusul setelah data penjualan dimuat
  try {
    google.script.run
      .withSuccessHandler(function(res) {
        var ttd = function() { return urutanTerlaris(menus).map(function(m) { return m.id; }).join(','); };
        var sebelum = ttd();   // urutan yang sedang tampil sebelum data penjualan masuk
        _populerIds = (res && Array.isArray(res.ids)) ? res.ids : [];
        _terjual = (res && res.terjual && typeof res.terjual === 'object') ? res.terjual : {};
        tambahTabPopuler();
        // Tampilkan ulang hanya bila urutan berubah, agar daftar tidak berkedip tanpa perlu
        if (currentKat === 'populer' || sebelum !== ttd()) renderMenu(currentKat);
      })
      .withFailureHandler(function() {
        if (!percobaan) setTimeout(function() { muatPopuler(1); }, 2500);   // coba sekali lagi bila gagal
      })
      .getProdukPopuler();
  } catch (e) {}
}

function tambahTabPopuler() {
  var wrap = document.getElementById('catWrap');
  if (!wrap) return;
  var lama = wrap.querySelector('[data-kat="populer"]');
  if (lama) lama.parentNode.removeChild(lama);
  // hanya produk yang masih ada di menu
  _populerIds = _populerIds.filter(function(id) { return menus.some(function(m) { return String(m.id) === String(id); }); });
  // Tab Populer selalu tampil; bila belum ada data penjualan isinya menampilkan pesan kosong.
  var btn = document.createElement('button');
  btn.className = 'cat-tab' + (currentKat === 'populer' ? ' active' : '');
  btn.setAttribute('data-kat', 'populer');
  btn.textContent = 'Populer';
  btn.onclick = function() { pilihKategori(btn, 'populer'); };
  var pertama = wrap.firstChild;   // tab "Semua"
  wrap.insertBefore(btn, pertama ? pertama.nextSibling : null);
}

function pilihKategori(btn, kat) {
  document.querySelectorAll('.cat-tab').forEach(function(t){ t.classList.remove('active'); });
  btn.classList.add('active');
  renderMenu(kat);
}

// ============ SLIDER FOTO PRODUK (geser kiri/kanan) ============
function sliderFotoHtml(fotos, alt) {
  var a = String(alt || '').replace(/"/g, '&quot;');
  var imgs = fotos.map(function(u, i) {
    return '<img' + (i === 0 ? ' class="on"' : '') + ' src="' + u + '" alt="' + a + '" loading="lazy" draggable="false" onerror="this.onerror=null;this.removeAttribute(\'src\');this.style.visibility=\'hidden\';">';
  }).join('');
  var dots = fotos.map(function(u, i) { return '<i' + (i === 0 ? ' class="on"' : '') + '></i>'; }).join('');
  return '<div class="pslider-box">' +
    '<div class="pslider">' + imgs + '</div>' +
    '<button type="button" class="pslider-nav prev" onclick="geserSlider(this,-1,event)">‹</button>' +
    '<button type="button" class="pslider-nav next" onclick="geserSlider(this,1,event)">›</button>' +
    '<div class="pslider-dots">' + dots + '</div>' +
  '</div>';
}
function geserSlider(btn, dir, ev) {
  if (ev) ev.stopPropagation();
  var s = btn.parentNode.querySelector('.pslider');
  if (s) setSlideFoto(s, idxSlideFoto(s) + dir);
}

// Ganti foto aktif slider (transisi muncul-dari-tengah diatur CSS) + perbarui titik indikator.
function idxSlideFoto(s) {
  var v = parseInt(s.getAttribute('data-idx'), 10);
  return isNaN(v) ? 0 : v;
}
function setSlideFoto(s, idx) {
  var imgs = s.querySelectorAll('img');
  if (!imgs.length) return;
  idx = Math.max(0, Math.min(imgs.length - 1, idx));
  for (var k = 0; k < imgs.length; k++) imgs[k].classList.toggle('on', k === idx);
  var dots = s.parentNode.querySelectorAll('.pslider-dots i');
  for (var d = 0; d < dots.length; d++) dots[d].classList.toggle('on', d === idx);
  s.setAttribute('data-idx', idx);
}

// Gestur geser (sentuh/mouse) di foto: geser kiri = foto berikutnya, geser kanan = sebelumnya.
// Setelah menggeser, klik yang menyusul diabaikan agar kartu tidak ikut terbuka.
var _swipeMulai = null, _swipeTs = 0;
document.addEventListener('pointerdown', function(e) {
  var s = e.target && e.target.closest ? e.target.closest('.pslider') : null;
  _swipeMulai = s ? { el: s, x: e.clientX, y: e.clientY } : null;
}, true);
document.addEventListener('pointerup', function(e) {
  var st = _swipeMulai; _swipeMulai = null;
  if (!st) return;
  var dx = e.clientX - st.x, dy = e.clientY - st.y;
  if (Math.abs(dx) > 35 && Math.abs(dx) > Math.abs(dy) * 1.2) {
    setSlideFoto(st.el, idxSlideFoto(st.el) + (dx < 0 ? 1 : -1));
    _swipeTs = Date.now();
  }
}, true);
document.addEventListener('pointercancel', function() { _swipeMulai = null; }, true);
document.addEventListener('click', function(e) {
  if (Date.now() - _swipeTs < 400 && e.target && e.target.closest && e.target.closest('.pslider')) {
    e.stopPropagation(); e.preventDefault();
  }
}, true);
// Perbarui titik indikator saat slider digeser (event scroll tidak bubble → pakai capture)
document.addEventListener('scroll', function(e) {
  var t = e.target;
  if (!t || !t.classList || !t.classList.contains('pslider')) return;
  var idx = Math.round(t.scrollLeft / (t.clientWidth || 1));
  var dots = t.parentNode.querySelectorAll('.pslider-dots i');
  for (var k = 0; k < dots.length; k++) dots[k].classList.toggle('on', k === idx);
}, true);

function renderMenu(kat) {
  currentKat = kat;
  var list = kat === 'semua' ? urutanTerlaris(menus)
    : kat === 'populer' ? _populerIds.map(function(id) { return menus.filter(function(m) { return String(m.id) === String(id); })[0]; }).filter(Boolean)
    : urutanTerlaris(menus.filter(function(m){ return m.kat === kat; }));
  var container = document.getElementById('menuContainer');
  if (!list.length) {
    var pesanKosong = kat === 'populer' ? 'Belum ada data penjualan untuk menentukan produk terlaris.' : 'Tidak ada menu di kategori ini.';
    container.innerHTML = '<div class="empty-state"><div class="ic">🍽️</div>' + pesanKosong + '</div>';
    document.getElementById('jmlProduk').textContent = '0 produk';
    return;
  }

  document.getElementById('jmlProduk').textContent = list.length + ' produk';
  container.innerHTML = list.map(function(m, i) {
    var idx = menus.indexOf(m);
    var imgHtml = (m.imgs && m.imgs.length > 1)
      ? sliderFotoHtml(m.imgs, m.nama)
      : m.img
      ? '<img src="' + m.img + '" alt="' + m.nama.replace(/"/g, '&quot;') + '" loading="lazy" onerror="this.onerror=null;this.style.display=\'none\';">'
      : '<div class="ph">🍽️</div>';
    var opsiBadge = m.opsiRaw ? '<span class="opsi-badge">Tersedia pilihan opsi</span>' : '';
    return '<div class="menu-card' + (!m.tersedia ? ' habis' : '') + '" onclick="bukaOpsiMenu(' + idx + ')">' +
      (!m.tersedia ? '<div class="habis-flag">Habis</div>' : '') +
      (kat === 'populer' ? '<div class="rank-flag">No. ' + (i + 1) + '</div>' : '') +
      '<div class="img-box">' + imgHtml + '</div>' +
      '<div class="info">' +
        kartuBrandHtml(m) +
        '<div class="nama">' + m.nama + '</div>' +
        opsiBadge +
        hargaDivHtml(m) +
      '</div>' + KARTU_BTN_HTML +
    '</div>';
  }).join('');
  animasiKartuMenu();
}

// ============ ANIMASI MASUK KARTU PRODUK ============
// Kartu muncul halus (fade + naik) ketika masuk layar, berurutan agar terasa mengalir.
// Kelas animasi dipasang lewat JS; bila animasi tidak didukung kartu tetap tampil normal.
var _animKartuObs = null;
function animasiKartuMenu() {
  if (_animKartuObs) { _animKartuObs.disconnect(); _animKartuObs = null; }
  var cards = document.querySelectorAll('#menuContainer .menu-card');
  if (!cards.length || typeof IntersectionObserver === 'undefined') return;
  if (window.matchMedia && window.matchMedia('(prefers-reduced-motion: reduce)').matches) return;
  Array.prototype.forEach.call(cards, function(c) { c.classList.add('anim'); });
  void document.getElementById('menuContainer').offsetWidth;   // pastikan keadaan awal terbaca sebelum transisi
  _animKartuObs = new IntersectionObserver(function(entries, obs) {
    var urut = 0;
    entries.forEach(function(en) {
      if (!en.isIntersecting) return;
      var c = en.target;
      obs.unobserve(c);
      var d = Math.min(urut++, 8) * 70;                       // jeda bertahap antar kartu (maks. ±0,5 dtk)
      c.style.setProperty('--d', d + 'ms');
      c.classList.add('in');
      // setelah selesai, kembalikan kartu ke gaya normal agar hover dll. tidak terpengaruh
      setTimeout(function() { c.classList.remove('anim', 'in'); c.style.removeProperty('--d'); }, d + 900);
    });
  }, { threshold: 0.1, rootMargin: '0px 0px -30px 0px' });
  Array.prototype.forEach.call(cards, function(c) { _animKartuObs.observe(c); });
}

// ============ DESAIN OPSI (gambar pilihan, mis. Kartu Ucapan) ============
// Diatur pemilik di Pengaturan → Desain Opsi. Dicocokkan ke opsi lewat NAMA OPSI (tidak peka huruf besar/kecil).
var _desainOpsi = {};   // { 'kartu ucapan': [{nama, url}] }
function muatDesainOpsi() {
  try {
    google.script.run
      .withSuccessHandler(function(res) {
        var map = {};
        ((res && res.list) || []).forEach(function(g) {
          var ds = (g.desain || []).filter(function(d) { return d && /^https:\/\//i.test(String(d.url || '')); });
          if (g && g.opsi && ds.length) map[String(g.opsi).trim().toLowerCase()] = ds;
        });
        _desainOpsi = map;
      })
      .withFailureHandler(function() {})
      .getDesainOpsiMenu();
  } catch (e) {}
}
function daftarDesainOpsi(namaOpsi) {
  return _desainOpsi[String(namaOpsi || '').trim().toLowerCase()] || [];
}
function pilihDesainOpsi(btn) {
  var box = btn.closest('.opt-note');
  if (!box) return;
  var sudah = btn.classList.contains('pilih');
  Array.prototype.forEach.call(box.querySelectorAll('.dsn-item'), function(b) { b.classList.remove('pilih'); });
  if (sudah) { box.removeAttribute('data-desain'); return; }   // klik lagi = batal pilih
  btn.classList.add('pilih');
  box.setAttribute('data-desain', btn.getAttribute('data-nama') || '');
}

function bukaZoomDesain(btn) {

function desainOpsiHtml(namaOpsi) {

// Radio: hanya opsi terpilih yang punya kolom catatan; checkbox: tampil saat dicentang, hilang saat dicabut.
function setCatatanOpsi(gi, ii, tampil) {
  var n = document.getElementById('optNote_' + gi + '_' + ii);
  if (!n) return;
  n.classList.toggle('show', !!tampil);
  var ta = n.querySelector('textarea');
  if (!tampil && ta) ta.value = '';
  if (!tampil) {
    n.removeAttribute('data-desain');
    Array.prototype.forEach.call(n.querySelectorAll('.dsn-item.pilih'), function(b) { b.classList.remove('pilih'); });
  }
}
function tampilCatatanOpsi(gi, ii, el) {
  // Ditunda satu tick: untuk checkbox, batas maksimal pilihan (pilihOpsi) bisa membatalkan centang.
  setTimeout(function() {
    if (!_pilihSaatIni) return;
    var g = parseOpsiMenu(_pilihSaatIni.opsiRaw)[gi];
    if (!g) return;
    if (g.type === 'radio') {
      for (var k = 0; k < g.items.length; k++) setCatatanOpsi(gi, k, k === ii && el.checked);
    } else {
      setCatatanOpsi(gi, ii, el.checked);
    }
  }, 0);
}
// Gabungkan catatan semua opsi yang sedang terpilih & terisi: "Nama Opsi: isi; Opsi Lain: isi".
function kumpulCatatanOpsi() {
  var out = [];
  var notes = document.querySelectorAll('#opsiGroupWrap .opt-note.show');
  for (var i = 0; i < notes.length; i++) {
    var ta = notes[i].querySelector('textarea');
    var v = ta ? ta.value.trim() : '';
    var dsn = notes[i].getAttribute('data-desain') || '';   // desain bergambar yang dipilih (jika ada)
    if (dsn) v = (v ? v + ' ' : '') + '(Desain: ' + dsn + ')';
    if (v) out.push(notes[i].getAttribute('data-nama') + ': ' + v);
  }
  return out.join('; ');
}

function pilihOpsi(gi, ii, el) {
  var g = parseOpsiMenu(_pilihSaatIni.opsiRaw)[gi];
  var item = g.items[ii];
  if (g.type === 'radio') {
    _opsiState[gi] = [item];
  } else {
    var current = _opsiState[gi] || [];
    if (el.checked) {
      if (current.length >= g.max) { el.checked = false; showToast('Maksimal pilih ' + g.max + ' untuk ' + g.group); return; }
      current.push(item);
    } else {
      current = current.filter(function(x){ return x.nama !== item.nama; });
    }
    _opsiState[gi] = current;
  }
  hitungTotalSheetOpsi();
}

var _crop = { img: null, url: null, S: 300, scale: 1, min: 1, max: 4, ox: 0, oy: 0, ptrs: {}, lastDist: 0 };
var _cropTotal = 0, _cropKe = 0;

function _cropTutup() {
  if (_crop.url) { try { URL.revokeObjectURL(_crop.url); } catch (e) {} }
  _crop.url = null; _crop.img = null; _crop.ptrs = {}; _crop.lastDist = 0;
}

function mulaiAntrianCrop() {
  _cropTutup();
  if (!_fotoAntrian.length || _fotoPelanggan.length >= FOTO_MAKS) {
    _fotoAntrian = [];
    tutupSheet('cropBg');
    renderFotoGrid();
    renderFotoRingkas();
    return;
  }
  var file = _fotoAntrian.shift();
  _cropKe++;
  if (!file.type || file.type.indexOf('image/') !== 0) {
    showToast('❌ ' + file.name + ' bukan file gambar');
    mulaiAntrianCrop();
    return;
  }
  if (file.size > FOTO_FILE_MAX) {
    showToast('❌ ' + file.name + ' terlalu besar (maks. 25 MB)');
    mulaiAntrianCrop();
    return;
  }
  var url = URL.createObjectURL(file);
  var img = new Image();
  img.onload = function() {
    _crop.url = url;
    _crop.img = img;
    bukaSheet('cropBg');
    document.getElementById('cropJudul').textContent = 'Atur Foto' + (_cropTotal > 1 ? ' (' + _cropKe + '/' + _cropTotal + ')' : '');
    var canvas = document.getElementById('cropCanvas');
    var S = Math.round(document.getElementById('cropStage').clientWidth) || 300;
    canvas.width = S; canvas.height = S;
    _crop.S = S;
    _crop.min = S / Math.min(img.naturalWidth, img.naturalHeight);
    _crop.max = _crop.min * 4;
    _crop.scale = _crop.min;
    _crop.ox = (S - img.naturalWidth * _crop.scale) / 2;
    _crop.oy = (S - img.naturalHeight * _crop.scale) / 2;
    document.getElementById('cropZoom').value = 0;
    gambarCrop();
  };
  img.onerror = function() {
    try { URL.revokeObjectURL(url); } catch (e) {}
    showToast('❌ ' + file.name + ' tidak dapat dibuka (format tidak didukung)');
    mulaiAntrianCrop();
  };
  img.src = url;
}

function _cropKlem() {
  var c = _crop;
  var w = c.img.naturalWidth * c.scale, h = c.img.naturalHeight * c.scale;
  c.ox = Math.min(0, Math.max(c.S - w, c.ox));
  c.oy = Math.min(0, Math.max(c.S - h, c.oy));
}

function gambarCrop() {
  var c = _crop;
  if (!c.img) return;
  _cropKlem();
  var canvas = document.getElementById('cropCanvas');
  var ctx = canvas.getContext('2d');
  ctx.fillStyle = '#fff';
  ctx.fillRect(0, 0, c.S, c.S);
  ctx.drawImage(c.img, c.ox, c.oy, c.img.naturalWidth * c.scale, c.img.naturalHeight * c.scale);
  var pxAsli = c.S / c.scale; // sisi area terpilih dalam piksel foto asli
  document.getElementById('cropWarn').textContent = pxAsli < 600 ? '⚠️ Resolusi foto rendah, hasil cetak bisa kurang tajam' : '';
}

function zoomCropKe(skalaBaru, cx, cy) {
  var c = _crop;
  if (!c.img) return;
  skalaBaru = Math.max(c.min, Math.min(c.max, skalaBaru));
  var rasio = skalaBaru / c.scale;
  c.ox = cx - (cx - c.ox) * rasio;
  c.oy = cy - (cy - c.oy) * rasio;
  c.scale = skalaBaru;
  document.getElementById('cropZoom').value = c.max > c.min ? Math.round((c.scale - c.min) / (c.max - c.min) * 100) : 0;
  gambarCrop();
}

function gunakanCrop() {
  var c = _crop;
  if (!c.img) return;
  var out = document.createElement('canvas');
  out.width = FOTO_OUT_PX; out.height = FOTO_OUT_PX;
  var k = FOTO_OUT_PX / c.S;
  var ctx = out.getContext('2d');
  ctx.fillStyle = '#fff';
  ctx.fillRect(0, 0, FOTO_OUT_PX, FOTO_OUT_PX);
  ctx.imageSmoothingQuality = 'high';
  ctx.drawImage(c.img, c.ox * k, c.oy * k, c.img.naturalWidth * c.scale * k, c.img.naturalHeight * c.scale * k);
  _fotoPelanggan.push(out.toDataURL('image/jpeg', 0.9));
  mulaiAntrianCrop();
}

function batalCrop() {
  _fotoAntrian = [];     // batal = berhenti untuk foto ini dan sisa antrian
  mulaiAntrianCrop();
}

(function pasangEventCrop() {
  var canvas = document.getElementById('cropCanvas');
  var zoom = document.getElementById('cropZoom');
  if (!canvas || !zoom) return;
  function skala() { var r = canvas.getBoundingClientRect(); return r.width ? _crop.S / r.width : 1; }
  function titik(e) { var r = canvas.getBoundingClientRect(); var k = skala(); return { x: (e.clientX - r.left) * k, y: (e.clientY - r.top) * k }; }

  canvas.addEventListener('pointerdown', function(e) {
    if (!_crop.img) return;
    try { canvas.setPointerCapture(e.pointerId); } catch (x) {}
    _crop.ptrs[e.pointerId] = { x: e.clientX, y: e.clientY };
    var ids = Object.keys(_crop.ptrs);
    if (ids.length === 2) {
      var a = _crop.ptrs[ids[0]], b = _crop.ptrs[ids[1]];
      _crop.lastDist = Math.hypot(a.x - b.x, a.y - b.y);
    }
  });
  canvas.addEventListener('pointermove', function(e) {
    var p = _crop.ptrs[e.pointerId];
    if (!p || !_crop.img) return;
    var ids = Object.keys(_crop.ptrs);
    if (ids.length === 1) {
      var k = skala();
      _crop.ox += (e.clientX - p.x) * k;
      _crop.oy += (e.clientY - p.y) * k;
      p.x = e.clientX; p.y = e.clientY;
      gambarCrop();
    } else if (ids.length >= 2) {
      p.x = e.clientX; p.y = e.clientY;
      var a = _crop.ptrs[ids[0]], b = _crop.ptrs[ids[1]];
      var dist = Math.hypot(a.x - b.x, a.y - b.y);
      if (_crop.lastDist > 0 && dist > 0) {
        var r = canvas.getBoundingClientRect(), kk = skala();
        var mx = ((a.x + b.x) / 2 - r.left) * kk, my = ((a.y + b.y) / 2 - r.top) * kk;
        zoomCropKe(_crop.scale * dist / _crop.lastDist, mx, my);
      }
      _crop.lastDist = dist;
    }
  });
  function lepas(e) {
    delete _crop.ptrs[e.pointerId];
    _crop.lastDist = 0;
    try { canvas.releasePointerCapture(e.pointerId); } catch (x) {}
  }
  canvas.addEventListener('pointerup', lepas);
  canvas.addEventListener('pointercancel', lepas);
  canvas.addEventListener('wheel', function(e) {
    if (!_crop.img) return;
    e.preventDefault();
    var t = titik(e);
    zoomCropKe(_crop.scale * (e.deltaY < 0 ? 1.08 : 0.92), t.x, t.y);
  }, { passive: false });
  zoom.addEventListener('input', function() {
    if (!_crop.img) return;
    var s = _crop.min + (_crop.max - _crop.min) * (Number(zoom.value) / 100);
    zoomCropKe(s, _crop.S / 2, _crop.S / 2);
  });
})();

// Upload semua foto di keranjang (berurutan) SEBELUM pesanan dicatat.
function unggahFotoPesanan(txId, btn, selesai, gagal) {
  var tugas = [];
  cart.forEach(function(c, ci) {
    (c.fotos || []).forEach(function(data, fi) { tugas.push({ c: c, itemKe: ci + 1, no: fi + 1, data: data }); });
  });
  if (!tugas.length) { selesai(); return; }
  var i = 0;
  function berikut() {
    if (i >= tugas.length) { selesai(); return; }
    var t = tugas[i];
    if (t.c._up && t.c._up[t.no]) { i++; berikut(); return; }   // sudah terunggah pada percobaan sebelumnya
    btn.textContent = '⏳ Mengupload foto ' + (i + 1) + '/' + tugas.length + '...';
    google.script.run
      .withSuccessHandler(function(res) {
        if (res && res.success) {
          t.c._up = t.c._up || {};
          t.c._up[t.no] = true;
          i++; berikut();
        } else {
          gagal((res && res.message) || 'Gagal mengupload foto');
        }
      })
      .withFailureHandler(function(err) { gagal(err && err.message ? err.message : 'Koneksi gagal'); })
      .uploadFotoPesanan({ orderId: txId, itemKe: t.itemKe, produk: t.c.nama, no: t.no, base64Data: t.data, mimeType: 'image/jpeg' });
  }
  berikut();
}

// ============ KERANJANG ============
function renderCartBar() {
  var bar = document.getElementById('cartBar');
  var totalQty = cart.reduce(function(s, c){ return s + c.qty; }, 0);
  var totalHarga = cart.reduce(function(s, c){ return s + c.hargaSatuan * c.qty; }, 0);
  document.getElementById('cartBadge').textContent = totalQty;
  document.getElementById('cartBarTotal').textContent = 'Rp ' + totalHarga.toLocaleString('id-ID');
  bar.classList.toggle('show', totalQty > 0);
  var hc = document.getElementById('headCartCnt');
  if (hc) { hc.textContent = totalQty; hc.classList.toggle('show', totalQty > 0); }
}

function bukaKeranjang() {
  isiDataPemesanTersimpan();   // nama & alamat dari pesanan sebelumnya (hanya mengisi kolom yang masih kosong)
  renderCartItems();
  bukaSheet('cartSheetBg');
}

function renderCartItems() {
  var wrap = document.getElementById('cartItemsWrap');
  if (!cart.length) {
    wrap.innerHTML = '<div class="empty-state" style="padding:24px 0;"><div class="ic">🛒</div>Keranjang masih kosong.</div>';
  } else {
    wrap.innerHTML = cart.map(function(c, i) {
      var optTxt = c.opsiTerpilih.length ? c.opsiTerpilih.map(function(o){ return o.nama; }).join(', ') : '';
      return '<div class="cart-item">' +
        '<div>' +
          '<div class="nm">' + c.qty + 'x ' + c.nama + '</div>' +
          (optTxt ? '<div class="opt">+ ' + optTxt + '</div>' : '') +
          (c.catatan ? '<div class="note">📝 ' + c.catatan + '</div>' : '') +
          (c.fotos && c.fotos.length ? '<div class="opt">📷 ' + c.fotos.length + ' foto untuk dicetak</div>' : '') +
          '<button class="rm" onclick="hapusDariKeranjang(' + i + ')">🗑 Hapus</button>' +
        '</div>' +
        '<div style="font-size:13px;font-weight:800;">Rp ' + (c.hargaSatuan * c.qty).toLocaleString('id-ID') + '</div>' +
      '</div>';
    }).join('');
  }
  var subtotal = cart.reduce(function(s, c){ return s + c.hargaSatuan * c.qty; }, 0);
  document.getElementById('sumSubtotal').textContent = 'Rp ' + subtotal.toLocaleString('id-ID');
  document.getElementById('sumTotal').textContent = 'Rp ' + subtotal.toLocaleString('id-ID');
  // Keterangan ongkos kirim dekat Total (sesuai pengaturan tiap produk di Manajemen Produk)
  var noteEl = document.getElementById('sumOngkirNote');
  if (noteEl) {
    var sudah = [], belum = [];
    cart.forEach(function(c) {
      var arr = c.ongkir === true ? sudah : belum;
      if (arr.indexOf(c.nama) < 0) arr.push(c.nama);
    });
    if (!cart.length) {
      noteEl.style.display = 'none';
    } else if (!belum.length) {
      noteEl.style.display = '';
      noteEl.style.color = 'var(--green, #2e7d32)';
      noteEl.textContent = '🚚 Sudah termasuk ongkos kirim';
    } else if (!sudah.length) {
      noteEl.style.display = '';
      noteEl.style.color = 'var(--muted)';
      noteEl.textContent = '🚚 Belum termasuk ongkos kirim';
    } else {
      noteEl.style.display = '';
      noteEl.style.color = 'var(--muted)';
      noteEl.textContent = '🚚 Sudah termasuk ongkos kirim: ' + sudah.join(', ') + ' · Belum termasuk ongkos kirim: ' + belum.join(', ');
    }
  }
}

function hapusDariKeranjang(i) {
  cart.splice(i, 1);
  resetUploadPending();
  renderCartItems();
  renderCartBar();
}

// ============ KIRIM PESANAN ============
// Memakai simpanTransaksi() — fungsi backend yang SAMA dipakai halaman
// Kasir — sehingga pesanan otomatis tercatat di Riwayat Transaksi &
// mengurangi stok produk yang kelola_stok = "Ya".
function prosesPesanan() {
  if (!cart.length) { showToast('Keranjang masih kosong'); return; }

  // Nama pemesan WAJIB — tanpa nama, pesanan tidak diteruskan.
  var inputNama = document.getElementById('namaPemesan');
  var namaPemesan = inputNama.value.trim();
  if (!namaPemesan) {
    inputNama.style.borderColor = 'var(--red)';
    inputNama.focus();
    showToast('⚠️ Nama pemesan wajib diisi');
    return;
  }

  // Alamat WAJIB (kecuali Patokan) — tanpa alamat lengkap, pesanan tidak diteruskan.
  var alamat = bacaAlamatPesanan();
  if (!alamat.ok) {
    _ALAMAT_FIELDS.forEach(function(f) {
      var el = document.getElementById(f.id);
      if (el && f.wajib && !alamat.v[f.id]) el.style.borderColor = 'var(--red)';
    });
    var elKosong = document.getElementById(alamat.kosong.id);
    if (elKosong) elKosong.focus();
    showToast('⚠️ ' + alamat.kosong.label + ' wajib diisi');
    return;
  }

  var btn = document.getElementById('btnPesanSekarang');
  btn.disabled = true; btn.textContent = '⏳ Mengirim pesanan...';

  var catatanUmum = document.getElementById('catatanUmum').value.trim();
  var subtotal = cart.reduce(function(s, c){ return s + c.hargaSatuan * c.qty; }, 0);

  var itemsPayload = cart.map(function(c) {
    var optTxt = c.opsiTerpilih.length ? ' (' + c.opsiTerpilih.map(function(o){ return o.nama; }).join(', ') + ')' : '';
    var noteTxt = c.catatan ? ' [' + c.catatan + ']' : '';
    return { id: c.menuId, name: c.nama + optTxt + noteTxt, cat: c.cat || '-', price: c.hargaSatuan, qty: c.qty, kelola_stok: c.kelolaStok || 'Ya' };
  });

  var fotoInfo = cart.filter(function(c) { return c.fotos && c.fotos.length; })
    .map(function(c) { return c.nama + ' (' + c.fotos.length + ' foto)'; }).join(', ');

  var catatanGabungan = [
    namaPemesan ? 'Pemesan: ' + namaPemesan : '',
    'Alamat: ' + alamatRingkas(alamat.v),
    fotoInfo ? 'Foto cetak: ' + fotoInfo : '',
    catatanUmum
  ].filter(Boolean).join(' | ');

  var txId = _pendingTxId || ('ONL-' + Date.now());
  _pendingTxId = txId;
  // Konfirmasi WhatsApp: rincian pesanan disiapkan & disalin SAAT tombol ditekan (agar clipboard diizinkan browser).
  _pesanWaTeks = ''; _pesanWaUrl = '';
  if (_linkWa) {
    _pesanWaTeks = buatTeksKonfirmasiWa(txId, namaPemesan, catatanUmum, subtotal, alamatBarisWa(alamat.v));
    _pesanWaUrl = buatUrlWa(_pesanWaTeks);
    salinTeks(_pesanWaTeks);
  }
  var payload = {
    id: txId,
    items: itemsPayload,
    total: subtotal,
    diskon: 0,
    pajak: 0,
    ongkir: 0,
    metodeBayar: 'Pesanan Menu Online',
    uangDiterima: 0,
    kembalian: 0,
    catatan: catatanGabungan || 'Pesanan via Menu Pelanggan',
    kasir: 'Menu Pelanggan',
    status: 'Menunggu Diproses'
  };

  var kirimTransaksiAkhir = function() {
  try {
    google.script.run
      .withSuccessHandler(function(res) {
        btn.disabled = false; btn.textContent = labelTombolPesan();
        if (res && res.success) {
          _pendingTxId = null;
          tutupSheet('cartSheetBg');
          document.getElementById('successMsg').textContent = 'Pesanan ' + txId + ' sudah diteruskan ke admin kami. Terimakasih untuk pesanannya.';
          bukaSheet('successSheetBg');
          tampilkanKonfirmasiWa();
          catatRiwayatPesanan(txId, namaPemesan, alamat.v, catatanUmum, subtotal);   // simpan ke riwayat (cache browser)
          cart = [];
          renderCartBar();
        } else {
          showToast('❌ ' + (res ? res.message : 'Gagal mengirim pesanan'));
        }
      })
      .withFailureHandler(function(err) {
        btn.disabled = false; btn.textContent = labelTombolPesan();
        showToast('❌ Error: ' + err.message);
      })
      .simpanTransaksi(payload);
  } catch (e) {
    btn.disabled = false; btn.textContent = labelTombolPesan();
    showToast('❌ Error: ' + e.message);
  }
  };

  // Foto pelanggan (jika ada) diunggah dulu; pesanan baru dicatat setelah semua foto tersimpan.
  unggahFotoPesanan(txId, btn, kirimTransaksiAkhir, function(msg) {
    btn.disabled = false; btn.textContent = labelTombolPesan();
    showToast('❌ ' + msg + ' — silakan tekan Pesan Sekarang lagi');
  });
}
